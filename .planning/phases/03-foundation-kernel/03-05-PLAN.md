    ```go
    func Sort(tasks map[string]Task) ([]Task, error) {
        const (white = 0; gray = 1; black = 2)
        color := make(map[string]int, len(tasks))
        order := make([]Task, 0, len(tasks))
        var dfs func(name string, path []string) error
        dfs = func(name string, path []string) error {
            t, ok := tasks[name]
            if !ok {
                return &coreerrors.ConfigError{
                    Key: "task.depends_on",
                    Message: "task " + name + " depends on unknown task — not registered",
                }
            }
            switch color[name] {
            case black: return nil
            case gray:
                cycle := append(path, name)
                return &coreerrors.ConfigError{
                    Key: "task.depends_on",
                    Message: "circular DependsOn detected: " + strings.Join(cycle, " → "),
                }
            }
            color[name] = gray
            for _, dep := range t.DependsOn() {
                if err := dfs(dep, append(path, name)); err != nil { return err }
            }
            color[name] = black
            order = append(order, t)
            return nil
        }
        names := make([]string, 0, len(tasks))
        for n := range tasks { names = append(names, n) }
        sort.Strings(names)
        for _, n := range names {
            if err := dfs(n, nil); err != nil { return nil, err }
        }
        return order, nil
    }
    ```

    **Sub-task F: `internal/core/scheduler/scheduler.go`** — main fork per ADR §7.2:

    ```go
    package scheduler

    import (
        "context"
        "encoding/json"
        "log/slog"
        "time"

        "golang.org/x/sync/errgroup"

        "github.com/six2dez/reconftw/internal/core/appctx"
        "github.com/six2dez/reconftw/internal/core/checkpoint"
        "github.com/six2dez/reconftw/internal/core/task"
    )

    type Scheduler struct {
        MaxConcurrent    int
        HeartbeatSeconds int
        PolicyOverride   map[string]FailurePolicy
        Checkpoint       checkpoint.Interface // per W15 — interface, not concrete *Store; allows MockCheckpoint substitution
        Log              *slog.Logger
    }

    func NewScheduler(maxJobs, heartbeatSec int, cp checkpoint.Interface, log *slog.Logger) *Scheduler {
        if maxJobs <= 0 { maxJobs = 4 }
        return &Scheduler{
            MaxConcurrent:    maxJobs,
            HeartbeatSeconds: heartbeatSec,
            Checkpoint:       cp,
            Log:              log,
        }
    }

    // runStage takes the FINAL *appctx.AppContext (no `any` placeholder).
    func (s *Scheduler) RunStage(ctx context.Context, module string, tasks []task.Task, app *appctx.AppContext) error {
        if len(tasks) == 0 { return nil }
        policy := policyFor(module, s.PolicyOverride)

        if policy == PolicyFailFast {
            g, gctx := errgroup.WithContext(ctx)
            g.SetLimit(s.MaxConcurrent)
            for _, t := range tasks {
                t := t
                g.Go(func() error { return s.runOne(gctx, t, app) })
            }
            return g.Wait()
        }

        // best_effort: zero-value errgroup — no peer cancellation
        g := new(errgroup.Group)
        g.SetLimit(s.MaxConcurrent)
        for _, t := range tasks {
            t := t
            g.Go(func() error {
                if err := s.runOne(ctx, t, app); err != nil {
                    s.Log.Warn("task_error_best_effort",
                        slog.String("task", t.Name()),
                        slog.String("module", t.Module()),
                        slog.Any("err", err))
                }
                return nil
            })
        }
        return g.Wait()
    }

    func (s *Scheduler) runOne(ctx context.Context, t task.Task, app *appctx.AppContext) error {
        cfgSliceJSON, _ := json.Marshal(app.Cfg)
        wordlistsLock := []byte("") // Phase 11 wires real wordlists.lock
        target := app.Target.Domain
        hash := checkpoint.InputHash(t.Name(), target, cfgSliceJSON, wordlistsLock)

        if s.Checkpoint != nil {
            done, err := s.Checkpoint.Done(ctx, t.Name(), target, hash)
            if err != nil { return err }
            if done {
                s.Log.Info("task_skipped_checkpoint_hit", slog.String("task", t.Name()))
                return nil
            }
            if err := s.Checkpoint.Begin(ctx, t.Name(), target, hash); err != nil { return err }
        }

        if la, ok := t.(task.LifecycleAware); ok {
            if err := la.OnStart(ctx, app); err != nil {
                s.Log.Warn("task_onstart_failed", slog.String("task", t.Name()), slog.Any("err", err))
            }
        }

        cadence := time.Duration(s.HeartbeatSeconds) * time.Second
        stopHB := startHeartbeat(ctx, t.Name(), t.Module(), cadence)
        defer stopHB()

        start := time.Now()
        result, runErr := t.Run(ctx, app)
        result.Duration = time.Since(start)

        if la, ok := t.(task.LifecycleAware); ok {
            if endErr := la.OnEnd(ctx, app, result); endErr != nil {
                s.Log.Warn("task_onend_failed", slog.String("task", t.Name()), slog.Any("err", endErr))
            }
        }

        if s.Checkpoint != nil {
            _ = s.Checkpoint.Complete(ctx, t.Name(), target, hash, result.Outputs, runErr)
        }
        return runErr
    }
    ```
