    **Sub-task C: `internal/core/task/topo.go`** — Kahn's algorithm OR DFS with WHITE/GRAY/BLACK marking:

    ```go
    package task

    import (
        coreerrors "github.com/six2dez/reconftw/internal/core/errors"
    )

    // Sort returns tasks in topological order. Detects cycles and missing dependencies.
    // Returns *errors.ConfigError on validation failure (per ADR §6 PITFALL NOTE).
    func Sort(tasks map[string]Task) ([]Task, error) {
        // 3-color DFS:
        //   white = unvisited
        //   gray  = in current recursion stack (back-edge = cycle)
        //   black = done
        const (white = 0; gray = 1; black = 2)
        color := make(map[string]int, len(tasks))
        order := make([]Task, 0, len(tasks))
        var dfs func(name string, path []string) error
        dfs = func(name string, path []string) error {
            t, ok := tasks[name]
            if !ok {
                return &coreerrors.ConfigError{
                    Key:     "task.depends_on",
                    Message: "task " + name + " depends on unknown task — not registered",
                }
            }
            switch color[name] {
            case black: return nil
            case gray:
                // Cycle: extract the cycle path
                cycle := append(path, name)
                return &coreerrors.ConfigError{
                    Key:     "task.depends_on",
                    Message: "circular DependsOn detected: " + strings.Join(cycle, " → "),
                }
            }
            color[name] = gray
            for _, dep := range t.DependsOn() {
                if err := dfs(dep, append(path, name)); err != nil { return err }
            }
            color[name] = black
            order = append(order, t)
            return nil
        }
        // Iterate in stable order
        names := make([]string, 0, len(tasks))
        for n := range tasks { names = append(names, n) }
        sort.Strings(names)
        for _, n := range names {
            if err := dfs(n, nil); err != nil { return nil, err }
        }
        return order, nil
    }
    ```

            Short: "Verify all tool binaries reachable + backend operational",
            RunE: func(cmd *cobra.Command, args []string) error {
                ctx := cmd.Context()
                printer := ui.NewPrinter(cmd.OutOrStdout(), ui.VerbosityNormal)
                failed := 0

                // Check 1: config parse
                start := time.Now()
                if cfg == nil {
                    printer.Status(ui.BadgeFAIL, "config.parse", time.Since(start))
                    failed++
                } else {
                    printer.Status(ui.BadgeOK, "config.parse", time.Since(start))
                }

                // Check 2: LocalBackend.HealthCheck
                start = time.Now()
                local := backend.NewLocalBackend(time.Duration(cfg.Concurrency.KillGraceSeconds) * time.Second)
                if err := local.HealthCheck(ctx); err != nil {
                    printer.Status(ui.BadgeFAIL, "backend.local", time.Since(start))
                    failed++
                } else {
                    printer.Status(ui.BadgeOK, "backend.local", time.Since(start))
                }

                // Check 3: ToolRegistry per-tool exec.LookPath
                tools := backend.Default.All()
                for _, t := range tools {
                    start = time.Now()
                    if _, err := exec.LookPath(t.Name); err != nil {
                        printer.Status(ui.BadgeFAIL, "tool."+t.Name, time.Since(start))
                        failed++
                    } else {
                        printer.Status(ui.BadgeOK, "tool."+t.Name, time.Since(start))
                    }
                }
                if len(tools) == 0 {
                    printer.Status(ui.BadgeOK, "tools (0 tools registered — Phase 3 baseline)", 0)
                }

                if failed > 0 {
                    return &exitCodeError{code: 1, msg: fmt.Sprintf("%d health checks failed", failed)}
                }
                return nil
            },
