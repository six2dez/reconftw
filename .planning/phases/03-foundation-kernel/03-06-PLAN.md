    ```go
    package main

    import (
        "context"
        "log/slog"
        "os"
        "os/signal"
        "syscall"

        "github.com/six2dez/reconftw/internal/core/appctx"
        "github.com/six2dez/reconftw/internal/core/config"
        "github.com/six2dez/reconftw/internal/core/log"
        "github.com/six2dez/reconftw/internal/core/output"
    )

    // Version, CommitSHA, BuildDate are set at build time via -ldflags="-X main.Version=v2.0.0 ..."
    var (
        Version   = "dev"
        CommitSHA = "unknown"
        BuildDate = "unknown"
    )

    func main() {
        if err := run(); err != nil {
            // exitCodeError carries a specific exit code (D-02 = 64); otherwise default 1.
            if ec, ok := err.(*exitCodeError); ok {
                if ec.msg != "" {
                    fmt.Fprintln(os.Stderr, ec.msg)
                }
                os.Exit(ec.code)
            }
            slog.Error("reconftw_exit_error", "err", err)
            os.Exit(1)
        }
    }

    func run() error {
        // STEP 1: signal.NotifyContext for SIGINT/SIGTERM
        ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
        defer stop()

        // STEP 2: Create the redactor FIRST (no log lines yet)
        redactor := &log.Redactor{}

        // STEP 3: Bootstrap logger BEFORE config load
        // (config load itself may emit slog lines via validation errors)
        bootstrapLogger := log.New(log.BootstrapLoggerConfig(), redactor)
        slog.SetDefault(bootstrapLogger)

        // STEP 4: Pre-cobra parseEarlyFlags (W14) — extract ONLY --config and --secrets
        // so logger+config can initialize before cobra owns full parsing.
        earlyFlags := parseEarlyFlags(os.Args[1:])

        // STEP 5: Load config from 8-source chain (CLI overrides come from cobra later)
        cfg, err := config.Load(config.LoadOptions{
            ConfigPath:  earlyFlags.configPath,
            SecretsPath: earlyFlags.secretsPath,
        })
        if err != nil {
            slog.Error("config_load_failed", "err", err)
            return err
        }

        // STEP 6: Rebuild logger from full config (level + format)
        logger := log.New(cfg.AsLoggerConfig(), redactor)
        slog.SetDefault(logger)

        // STEP 7: Register all Secret field values BEFORE AppContext.Boot
        registerSecrets(cfg, redactor)

        // STEP 8: Boot AppContext if a target is set (otherwise version/health-check work without one)
        var app *appctx.AppContext
        if cfg.Target != "" {
            target, err := appctx.NewTarget(cfg.Target, cfg.Scope, "")
            if err != nil { return err }
            workdir, err := output.WorkspaceInit(cfg.Output.WorkspacesRoot, target.Domain)
            if err != nil { return err }
            target.WorkDir = workdir
            app, err = appctx.Boot(ctx, logger, cfg, target)
            if err != nil { return err }
        }

        // STEP 9: Execute cobra root command
        rootCmd := newRootCmd(app, redactor, cfg)
        return rootCmd.ExecuteContext(ctx)
    }

    type earlyFlagSet struct {
        configPath  string
        secretsPath string
    }

    // parseEarlyFlags scans args for ONLY --config FILE and --secrets FILE per W14.
    // All other flags are owned by cobra and parsed inside Execute().
    // Behavior: accepts both `--config=PATH` and `--config PATH` forms.
    func parseEarlyFlags(args []string) earlyFlagSet {
        var efs earlyFlagSet
        for i := 0; i < len(args); i++ {
            a := args[i]
            switch {
            case a == "--config" && i+1 < len(args):
                efs.configPath = args[i+1]
                i++
            case strings.HasPrefix(a, "--config="):
                efs.configPath = strings.TrimPrefix(a, "--config=")
            case a == "--secrets" && i+1 < len(args):
                efs.secretsPath = args[i+1]
                i++
            case strings.HasPrefix(a, "--secrets="):
                efs.secretsPath = strings.TrimPrefix(a, "--secrets=")
            }
        }
        return efs
    }

    // registerSecrets registers all log.Secret field values from cfg with the redactor.
    // CRITICAL: MUST be called immediately after config load, before AppContext.Boot.
    // The full list of secret fields tracks ADR §2.2 + §10.1 per W10:
    //   Notifications.Slack.WebhookURL
    //   Notifications.Telegram.BotToken
    //   Notifications.Discord.WebhookURL
    //   AI.OpenAIKey
    //   AI.AnthropicKey
    //   MCP.APIKey
    //   Axiom.SSHKey (when present in ADR §2.2 Axiom block)
    func registerSecrets(cfg *config.Config, r *log.Redactor) {
        r.Register(string(cfg.Notifications.Slack.WebhookURL))
        r.Register(string(cfg.Notifications.Telegram.BotToken))
        r.Register(string(cfg.Notifications.Discord.WebhookURL))
        r.Register(string(cfg.AI.OpenAIKey))
        r.Register(string(cfg.AI.AnthropicKey))
        r.Register(string(cfg.MCP.APIKey))
        // Axiom credentials per ADR §2.2 — included when the config struct exposes them as log.Secret
        if cfg.Axiom.SSHKey != "" {
            r.Register(string(cfg.Axiom.SSHKey))
        }
    }
    ```

    func newHealthCheckCmd(app *appctx.AppContext, cfg *config.Config) *cobra.Command {
        return &cobra.Command{
            Use:   "health-check",
            Short: "Verify all tool binaries reachable + backend operational",
            RunE: func(cmd *cobra.Command, args []string) error {
                ctx := cmd.Context()
                printer := ui.NewPrinter(cmd.OutOrStdout(), ui.VerbosityNormal)
                criticalFail := false

                // Check 1: config parse
                start := time.Now()
                if cfg == nil {
                    printer.Status(ui.BadgeFAIL, "config.parse", time.Since(start))
                    criticalFail = true
                } else {
                    printer.Status(ui.BadgeOK, "config.parse", time.Since(start))
                }

                // Check 2: LocalBackend.HealthCheck
                start = time.Now()
                local := backend.NewLocalBackend(time.Duration(cfg.Concurrency.KillGraceSeconds) * time.Second)
                if err := local.HealthCheck(ctx); err != nil {
                    printer.Status(ui.BadgeFAIL, "backend.local", time.Since(start))
                    criticalFail = true
                } else {
                    printer.Status(ui.BadgeOK, "backend.local", time.Since(start))
                }

                // Check 3: ToolRegistry per-tool exec.LookPath + Critical-tier check (Blocker 5)
                _ = backend.Default.Discover(ctx)
                missingCritical := backend.Default.MissingCritical()
                missingRequired := backend.Default.MissingRequired()
                missingCritSet := map[string]bool{}
                for _, n := range missingCritical { missingCritSet[n] = true }
                missingReqSet := map[string]bool{}
                for _, n := range missingRequired { missingReqSet[n] = true }

                for _, t := range backend.Default.All() {
                    start = time.Now()
                    if missingCritSet[t.Name] {
                        // Missing-and-critical → FAIL
                        printer.Status(ui.BadgeFAIL, "tool."+t.Name+" (critical)", time.Since(start))
                        criticalFail = true
                    } else if missingReqSet[t.Name] {
                        // Missing-but-required (not critical) → WARN, no exit failure
                        printer.Status(ui.BadgeWARN, "tool."+t.Name+" (required)", time.Since(start))
                    } else {
                        printer.Status(ui.BadgeOK, "tool."+t.Name, time.Since(start))
                    }
                }
                if len(backend.Default.All()) == 0 {
                    printer.Status(ui.BadgeOK, "tools (0 tools registered — Phase 3 baseline)", 0)
                }

                if criticalFail {
                    return &exitCodeError{code: 1, msg: fmt.Sprintf("%d critical health checks failed", len(missingCritical))}
                }
                return nil
            },
        }
    }
