    package testutil

    import (
        "context"
        "os"
        "path/filepath"
        "sync"
        "time"

        coreerrors "github.com/six2dez/reconftw/internal/core/errors"
        "github.com/six2dez/reconftw/internal/core/backend"
    )

    // MockBackend is a deterministic Backend used in unit + integration tests.
    // Reads tool output from internal/core/testutil/fixtures/<tool>/<scenario>.txt.
    // Never spawns a subprocess.
    type MockBackend struct {
        mu              sync.RWMutex
        FixturesDir     string
        Scenario        string
        exitCode        int
        runError        error
        healthCheckErr  error
        capacity        int
    }

    func NewMockBackend(fixturesDir string) *MockBackend {
        return &MockBackend{
            FixturesDir: fixturesDir,
            Scenario:    "default",
            capacity:    4,
        }
    }

    func (m *MockBackend) SetScenario(s string) { m.mu.Lock(); defer m.mu.Unlock(); m.Scenario = s }
    func (m *MockBackend) SetExitCode(c int) { m.mu.Lock(); defer m.mu.Unlock(); m.exitCode = c }
    func (m *MockBackend) SetError(err error) { m.mu.Lock(); defer m.mu.Unlock(); m.runError = err }
    func (m *MockBackend) SetHealthCheckError(err error) { m.mu.Lock(); defer m.mu.Unlock(); m.healthCheckErr = err }
    func (m *MockBackend) SetCapacity(c int) { m.mu.Lock(); defer m.mu.Unlock(); m.capacity = c }

    func (m *MockBackend) Exec(ctx context.Context, t *backend.Tool, args []string) (*backend.Result, error) {
        m.mu.RLock()
        scenario := m.Scenario
        exitCode := m.exitCode
        runErr := m.runError
        m.mu.RUnlock()
        if runErr != nil {
            return nil, runErr
        }
        path := filepath.Join(m.FixturesDir, t.Name, scenario+".txt")
        data, err := os.ReadFile(path)
        if err != nil {
            if os.IsNotExist(err) {
                // Try .jsonl extension as fallback
                alt := filepath.Join(m.FixturesDir, t.Name, scenario+".jsonl")
                if d, e := os.ReadFile(alt); e == nil { data = d; err = nil }
            }
            if err != nil {
                return nil, &coreerrors.ToolError{
                    Tool:     t.Name,
                    ExitCode: -1,
                    Inner:    err,
                }
            }
        }
        if exitCode != 0 {
            return &backend.Result{Stdout: data, ExitCode: exitCode, Duration: time.Millisecond},
                &coreerrors.ToolError{Tool: t.Name, ExitCode: exitCode}
        }
        return &backend.Result{Stdout: data, ExitCode: 0, Duration: time.Millisecond}, nil
    }

    func (m *MockBackend) Stream(ctx context.Context, t *backend.Tool, args []string) (<-chan backend.Event, error) {
        m.mu.RLock()
        scenario := m.Scenario
        runErr := m.runError
        m.mu.RUnlock()
        if runErr != nil { return nil, runErr }
        path := filepath.Join(m.FixturesDir, t.Name, scenario+".txt")
        f, err := os.Open(path)
        if err != nil {
            // try .jsonl
            altPath := filepath.Join(m.FixturesDir, t.Name, scenario+".jsonl")
            f, err = os.Open(altPath)
            if err != nil { return nil, &coreerrors.ToolError{Tool: t.Name, Inner: err} }
        }
        ch := make(chan backend.Event, 16)
        go func() {
            defer close(ch)
            defer f.Close()
            scanner := bufio.NewScanner(f)
            scanner.Buffer(make([]byte, 1024*1024), 10*1024*1024)
            for scanner.Scan() {
                select {
                case <-ctx.Done():
                    return
                case ch <- backend.Event{Line: append([]byte(nil), scanner.Bytes()...), Source: t.Name}:
                }
            }
        }()
        return ch, nil
    }

    func (m *MockBackend) HealthCheck(ctx context.Context) error {
        m.mu.RLock()
        defer m.mu.RUnlock()
        return m.healthCheckErr
    }

    func (m *MockBackend) Capacity() int {
        m.mu.RLock()
        defer m.mu.RUnlock()
        return m.capacity
    }

    package testutil

    import (
        "context"
        "sort"
        "strings"

        "time"

    // MockCheckpoint is an in-memory checkpoint store satisfying the *checkpoint.Store API.
    // Used in unit tests to avoid SQLite I/O.
    type MockCheckpoint struct {
        mu      sync.RWMutex
        records map[string]CheckpointRecord
    }

    type CheckpointRecord struct {
        TaskName    string
        Target      string
        InputHash   string
        Status      string  // "in-progress", "done", "errored", "cancelled", "skipped"
        StartedAt   time.Time
        FinishedAt  *time.Time
        OutputPaths []string
        ErrorClass  string
    }

    func NewMockCheckpoint() *MockCheckpoint {
        return &MockCheckpoint{records: map[string]CheckpointRecord{}}
    }

    func (m *MockCheckpoint) key(taskName, target, hash string) string {
        return taskName + "\x00" + target + "\x00" + hash
    }

    func (m *MockCheckpoint) Begin(ctx context.Context, taskName, target, inputHash string) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        m.records[m.key(taskName, target, inputHash)] = CheckpointRecord{
            TaskName: taskName, Target: target, InputHash: inputHash,
            Status: "in-progress", StartedAt: time.Now(),
        }
        return nil
    }

    func (m *MockCheckpoint) Done(ctx context.Context, taskName, target, inputHash string) (bool, error) {
        m.mu.RLock()
        defer m.mu.RUnlock()
        r, ok := m.records[m.key(taskName, target, inputHash)]
        if !ok { return false, nil }
        return r.Status == "done", nil
    }

    func (m *MockCheckpoint) Complete(ctx context.Context, taskName, target, inputHash string, outputPaths []string, runErr error) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        r := m.records[m.key(taskName, target, inputHash)]
        now := time.Now()
        r.FinishedAt = &now
        r.OutputPaths = outputPaths
        if runErr != nil {
            r.Status = "errored"
            r.ErrorClass = classifyMockErr(runErr)
        } else {
            r.Status = "done"
        }
        m.records[m.key(taskName, target, inputHash)] = r
        return nil
    }

    func (m *MockCheckpoint) AllRecords() []CheckpointRecord {
        m.mu.RLock()
        defer m.mu.RUnlock()
        out := make([]CheckpointRecord, 0, len(m.records))
        for _, r := range m.records { out = append(out, r) }
        sort.Slice(out, func(i, j int) bool {
            if out[i].TaskName != out[j].TaskName { return out[i].TaskName < out[j].TaskName }
            return out[i].Target < out[j].Target
        })
        return out
    }

    func (m *MockCheckpoint) Reset() {
        m.mu.Lock()
        defer m.mu.Unlock()
        m.records = map[string]CheckpointRecord{}
    }

    func classifyMockErr(err error) string {
        // Simplified — production checkpoint uses errors.As across 7 typed structs
        return err.Error()
    }

    // MockOutputTree is an in-memory OutputTree for tests. Append stores lines per artefact;
    // Lines(artefact) returns them in append order for assertions.
    type MockOutputTree struct {
        mu       sync.RWMutex
        appends  map[string][][]byte
    }

    func NewMockOutputTree() *MockOutputTree {
        return &MockOutputTree{appends: map[string][][]byte{}}
    }

    func (m *MockOutputTree) Append(artefact string, lines [][]byte) error {
        m.mu.Lock()
        defer m.mu.Unlock()
        cp := make([][]byte, len(lines))
        for i, l := range lines { cp[i] = append([]byte(nil), l...) }
        m.appends[artefact] = append(m.appends[artefact], cp...)
        return nil
    }

    func (m *MockOutputTree) Lines(artefact string) []string {
        m.mu.RLock()
        defer m.mu.RUnlock()
        ls := m.appends[artefact]
        out := make([]string, len(ls))
        for i, l := range ls { out[i] = string(l) }
        return out
    }

    func (m *MockOutputTree) Reset() {
        m.mu.Lock()
        defer m.mu.Unlock()
        m.appends = map[string][][]byte{}
    }

        Tools         []struct {
            Name           string   `toml:"name"`
            Kind           string   `toml:"kind"`
            GoModule       string   `toml:"go_module"`
            Description    string   `toml:"description"`
            DefaultArgs    []string `toml:"default_args"`
            TimeoutSeconds int      `toml:"timeout_seconds"`
        } `toml:"tools"`
    }

    func init() {
        var lock toolsLockSchema
        if err := tomlv2.Unmarshal(toolsLockData, &lock); err != nil {
            // Don't panic at init — log and continue with empty registry.
            slog.Error("tools_lock_parse_failed", "err", err)
            return
        }
        for _, t := range lock.Tools {
            tool := &Tool{
                Name:        t.Name,
                DefaultArgs: append([]string(nil), t.DefaultArgs...),
                Timeout:     time.Duration(t.TimeoutSeconds) * time.Second,
            }
            // Path resolved later via Discover/exec.LookPath
            Default.Register(tool)
        }
        slog.Info("tools_lock_loaded", "tool_count", len(lock.Tools))
    }
