    ```go
    package output

    import (
        "net/url"
        "regexp"
        "strings"
    )

    type ScopeFilter interface {
        IsInScope(host string) bool
        IsInScopeURL(rawurl string) bool
    }

    type DefaultScopeFilter struct {
        Patterns []string  // e.g. ["*.example.com", "example.com"]
    }

    var domainRe = regexp.MustCompile(`^[a-zA-Z0-9.-]+$`)  // REUSE from spike main.go:29 + lib/validation.sh

    func (f *DefaultScopeFilter) IsInScope(host string) bool {
        host = strings.ToLower(strings.TrimSpace(host))
        if !domainRe.MatchString(host) { return false }
        for _, pat := range f.Patterns {
            if matchScopePattern(host, pat) { return true }
        }
        return false
    }

    // matchScopePattern is the canonical scope check.
    // Mirrors lib/validation.sh:is_in_scope_host (anchored hostname; no substring false positives).
    func matchScopePattern(host, pat string) bool {
        pat = strings.ToLower(strings.TrimSpace(pat))
        if strings.HasPrefix(pat, "*.") {
            base := strings.TrimPrefix(pat, "*.")
            return host == base || strings.HasSuffix(host, "."+base)  // anchored — prevents "examplecom.evil.org" false positive
        }
        return host == pat
    }

    func (f *DefaultScopeFilter) IsInScopeURL(rawurl string) bool {
        u, err := url.Parse(rawurl)
        if err != nil { return false }
        return f.IsInScope(u.Hostname())
    }
    ```

    ```go
    package checkpoint

    import (
        "crypto/sha256"
        "encoding/hex"
    )

    // InputHash computes the SHA-256 of (config slice JSON + wordlists.lock content).
    // Changing the relevant config slice OR the wordlists.lock invalidates the checkpoint
    // and forces re-run on the next startup. Per ADR §3.4 line 1319.
    func InputHash(taskName string, target string, cfgSliceJSON []byte, wordlistsLockContent []byte) string {
        h := sha256.New()
        h.Write([]byte(taskName))   // task_name binding
        h.Write([]byte{0})           // null separator (prevents prefix collisions)
        h.Write([]byte(target))      // target binding
        h.Write([]byte{0})
        h.Write(cfgSliceJSON)
        h.Write([]byte{0})
        h.Write(wordlistsLockContent)
        return hex.EncodeToString(h.Sum(nil))
    }
    ```

    Note: ADR specifies `SHA-256(config_slice_json + wordlists.lock_content)` — task_name + target are already in the PRIMARY KEY, so technically they're not part of input_hash per ADR formula. However, including them prevents accidental matches across different tasks/targets with same config. Reconcile: ADR's PRIMARY KEY is `(task_name, target, input_hash)` — the input_hash uniqueness is per-(task,target). Include task+target in hash bytes anyway as belt-and-suspenders (matches reasonable security expectations).

    ```go
        StatusErrored    Status = "errored"
        StatusCancelled  Status = "cancelled"
        StatusSkipped    Status = "skipped"

        OutputPaths []string

        db, err := sql.Open("sqlite", path+"?_pragma=journal_mode(WAL)&_pragma=foreign_keys(1)")
        if err != nil { return nil, err }

    // Begin inserts an in-progress row. Uses BEGIN IMMEDIATE per ADR §3.4.
    func (s *Store) Begin(ctx context.Context, taskName, target, inputHash string) error {
        tx, err := s.db.BeginTx(ctx, nil)  // BEGIN IMMEDIATE — modernc/sqlite supports via PRAGMA
        if err != nil { return err }
        defer tx.Rollback()
        _, err = tx.ExecContext(ctx, `
            INSERT OR REPLACE INTO tasks (task_name, target, input_hash, status, started_at)
            VALUES (?, ?, ?, ?, ?)
        `, taskName, target, inputHash, string(StatusInProgress), time.Now().UTC().Format(time.RFC3339))

        return tx.Commit()
    }

    // Done returns true if a row with status=done exists for (taskName, target, inputHash).
    func (s *Store) Done(ctx context.Context, taskName, target, inputHash string) (bool, error) {
        var status string
        err := s.db.QueryRowContext(ctx, `
            SELECT status FROM tasks WHERE task_name = ? AND target = ? AND input_hash = ?
        `, taskName, target, inputHash).Scan(&status)
        if err == sql.ErrNoRows { return false, nil }
        if err != nil { return false, err }
        return status == string(StatusDone), nil
    }

    // Complete updates the row to a terminal status (done/errored/cancelled/skipped).
    // err is examined to extract error_class via errors.As; nil err → empty error_class + StatusDone.
    func (s *Store) Complete(ctx context.Context, taskName, target, inputHash string, outputPaths []string, runErr error) error {
        status := StatusDone
        errorClass := ""
        if runErr != nil {
            status = StatusErrored
            errorClass = classifyError(runErr)
        }
        outputs := strings.Join(outputPaths, ",")
        // Look up started_at to compute duration
        var startedAt string
        if err := s.db.QueryRowContext(ctx, `SELECT started_at FROM tasks WHERE task_name=? AND target=? AND input_hash=?`, taskName, target, inputHash).Scan(&startedAt); err != nil {
            // No row? Just use 0 duration
            startedAt = ""
        }
        durationMs := int64(0)
        if startedAt != "" {
            if t, err := time.Parse(time.RFC3339, startedAt); err == nil {
                durationMs = time.Since(t).Milliseconds()
            }
        }
        _, err := s.db.ExecContext(ctx, `
            UPDATE tasks SET status = ?, finished_at = ?, duration_ms = ?, output_paths = ?, error_class = ?
            WHERE task_name = ? AND target = ? AND input_hash = ?
        `, string(status), time.Now().UTC().Format(time.RFC3339), durationMs, outputs, errorClass, taskName, target, inputHash)
        return err
    }

    // Close closes the underlying DB connection.
    func (s *Store) Close() error { return s.db.Close() }

    // classifyError returns the typed error name from the 7-class hierarchy.

            ce *coreerrors.ConfigError
            se *coreerrors.ScopeError
            cm *coreerrors.ChecksumMismatch

        switch {
        case coreerrors.As(err, &te): return "ToolError"
        case coreerrors.As(err, &tt): return "ToolTimeout"
        case coreerrors.As(err, &oos): return "OutOfScope"
        case coreerrors.As(err, &ae): return "AxiomFailure"
        case coreerrors.As(err, &ce): return "ConfigError"
        case coreerrors.As(err, &se): return "ScopeError"
        case coreerrors.As(err, &cm): return "ChecksumMismatch"
        default: return "Unknown"
        }
    }
    ```
