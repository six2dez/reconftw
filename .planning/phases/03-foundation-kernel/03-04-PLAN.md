    `(*LocalBackend) Exec`: buffered output; lift kill-tree pattern from spike `proc.go`:
    1. `cmd := exec.CommandContext(ctx, t.Path, args...)`
    2. `cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}` (process group isolation)
    3. `cmd.WaitDelay = b.KillGrace` (default 5s)
    4. `cmd.Cancel = func() error { if cmd.Process == nil { return nil }; return syscall.Kill(-cmd.Process.Pid, syscall.SIGTERM) }`
    5. Capture stdout/stderr via `cmd.StdoutPipe()` / `cmd.StderrPipe()` into bytes.Buffer
    6. `cmd.Start()`
    7. Launch group-SIGKILL goroutine — when ctx cancelled, sleep `WaitDelay + 500ms`, then `syscall.Kill(-pgid, syscall.SIGKILL)` (ADR-cited workaround for Go's WaitDelay-only-kills-direct-child bug)
    8. `cmd.Wait()` and inspect ExitCode
    9. Return `*Result` on success; `*coreerrors.ToolError{Tool: t.Name, ExitCode: ..., Stderr: lastKB(stderrBuf), Inner: runErr}` on non-zero exit
    10. Return `*coreerrors.ToolTimeout{Tool: t.Name, Timeout: ...}` if ctx deadline exceeded (`errors.Is(ctx.Err(), context.DeadlineExceeded)`)

    `(*LocalBackend) Stream`: streamed output; same kill-tree as Exec but yields Events on a channel:
    1-7. Same as Exec but use both StdoutPipe and StderrPipe
    8. Spawn 2 goroutines (stdout reader + stderr reader) each scanning lines and sending Events on the channel
    9. Use `bufio.NewScanner` with `Buffer(make([]byte, 1024*1024), 10*1024*1024)` (1MiB initial / 10MiB max per RESEARCH.md §Pattern 3 — required for httpx 10M-line JSON)
    10. Close the channel when both readers AND `cmd.Wait()` complete
    11. CRITICAL: caller MUST drain the channel until closed (per ADR §5.2 line 1674-1675); document in godoc

    Helper functions:
    - `lastKB(buf bytes.Buffer) string` — returns last 1024 bytes of buf (per ADR §6 line 1806 stderr truncation)

    ```go
    type ToolRegistry struct {
        mu       sync.RWMutex
        tools    map[string]*Tool
        missing  []string  // populated by Discover
    }

    var Default = &ToolRegistry{tools: map[string]*Tool{}}

    func (r *ToolRegistry) Register(t *Tool) {
        r.mu.Lock()
        defer r.mu.Unlock()
        if _, exists := r.tools[t.Name]; exists {
            panic("reconftw: duplicate tool registration: " + t.Name)
        }
        r.tools[t.Name] = t
    }

    func (r *ToolRegistry) Lookup(name string) (*Tool, bool) {
        r.mu.RLock()
        defer r.mu.RUnlock()
        t, ok := r.tools[name]
        return t, ok
    }

    func (r *ToolRegistry) All() []*Tool {
        r.mu.RLock()
        defer r.mu.RUnlock()
        names := make([]string, 0, len(r.tools))
        for n := range r.tools { names = append(names, n) }
        sort.Strings(names)
        out := make([]*Tool, 0, len(names))
        for _, n := range names { out = append(out, r.tools[n]) }
        return out
    }

    // Discover walks PATH for each registered tool via exec.LookPath; populates .Path.
    // Tools not found are recorded in r.missing for later Health-Check.
    // Phase 3 ships this minimal mechanism; Phase 4-7 add criticality tagging.
    func (r *ToolRegistry) Discover(ctx context.Context) error {
        r.mu.Lock()
        defer r.mu.Unlock()
        r.missing = nil
        for name, t := range r.tools {
            path, err := exec.LookPath(name)
            if err != nil {
                r.missing = append(r.missing, name)
                continue
            }
            t.Path = path
        }
        return nil
    }

    // MissingRequired returns the list of registered tools not found on PATH.
    // Phase 3: all registered tools are treated as required.
    // Phase 4-7: extend with criticality tagging (some tools optional).
    func (r *ToolRegistry) MissingRequired() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()
        return append([]string(nil), r.missing...)
    }
    ```

    ```go
    package backend

    import (
        "context"
        "sync"

        "golang.org/x/time/rate"
    )

    type RateLimiter struct {
        mu          sync.RWMutex
        perTool     map[string]*rate.Limiter
        global      *rate.Limiter
    }

    func NewRateLimiter(perToolRPS map[string]int, globalRPS int) *RateLimiter {
        rl := &RateLimiter{perTool: make(map[string]*rate.Limiter, len(perToolRPS))}
        for tool, rps := range perToolRPS {
            if rps > 0 {
                rl.perTool[tool] = rate.NewLimiter(rate.Limit(rps), rps)  // burst = 1s budget
            }
        }
        if globalRPS > 0 {
            rl.global = rate.NewLimiter(rate.Limit(globalRPS), globalRPS)
        }
        return rl
    }

    // Wait blocks until both per-tool and global limits allow proceeding.
    // For an unknown tool with no global limit: returns immediately.
    func (rl *RateLimiter) Wait(ctx context.Context, tool string) error {
        rl.mu.RLock()
        toolLim := rl.perTool[tool]
        global := rl.global
        rl.mu.RUnlock()
        if toolLim != nil {
            if err := toolLim.Wait(ctx); err != nil { return err }
        }
        if global != nil {
            if err := global.Wait(ctx); err != nil { return err }
        }
        return nil
    }
    ```

    ```go
    package scheduler

    import (
        "context"
        "log/slog"
        "time"
    )

    // startHeartbeat launches a goroutine that emits "task_heartbeat" log events at cadence.
    // Returns a cancel function to stop the heartbeat when the task completes.
    // The first heartbeat fires at 2*cadence (avoid spam for short tasks per XCUT-09 spec).
    func startHeartbeat(ctx context.Context, taskName, module string, cadence time.Duration) (cancel func()) {
        if cadence <= 0 {
            return func() {}
        }
        ctx, cancelFn := context.WithCancel(ctx)
        go func() {
            start := time.Now()
            ticker := time.NewTicker(cadence)
            defer ticker.Stop()
            // Skip the first 2*cadence (don't heartbeat short tasks)
            firstTick := time.NewTimer(2 * cadence)
            defer firstTick.Stop()
            select {
            case <-ctx.Done():
                return
            case <-firstTick.C:
            }
            for {
                select {
                case <-ctx.Done():
                    return
                case <-ticker.C:
                    slog.Info("task_heartbeat",
                        slog.String("task", taskName),
                        slog.String("module", module),
                        slog.Duration("elapsed", time.Since(start)),
                    )
                }
            }
        }()
        return cancelFn
    }
    ```

    ```go
    type Scheduler struct {
        MaxConcurrent      int
        HeartbeatSeconds   int
        PolicyOverride     map[string]FailurePolicy
        Checkpoint         *checkpoint.Store
        Log                *slog.Logger
        // Plan 05 wires: Runner, OutputTree, Target, AppContext
    }

    func NewScheduler(maxJobs, heartbeatSec int, cp *checkpoint.Store, log *slog.Logger) *Scheduler {
        if maxJobs <= 0 { maxJobs = 4 }
        return &Scheduler{
            MaxConcurrent:    maxJobs,
            HeartbeatSeconds: heartbeatSec,
            Checkpoint:       cp,
            Log:              log,
        }
    }

    // runStage runs all tasks in a single module group in parallel up to MaxConcurrent,
    // applying the configured failure_policy for the module (fail_fast vs best_effort).
    // Lift from ADR §7.2 lines 1996-2050 verbatim.
    func (s *Scheduler) runStage(ctx context.Context, module string, tasks []task.Task, app any) error {
        if len(tasks) == 0 { return nil }
        policy := policyFor(module, s.PolicyOverride)

        if policy == PolicyFailFast {
            g, gctx := errgroup.WithContext(ctx)
            g.SetLimit(s.MaxConcurrent)
            for _, t := range tasks {
                t := t
                g.Go(func() error { return s.runOne(gctx, t, app) })
            }
            return g.Wait()
        }

        // best_effort: zero-value errgroup — no peer cancellation
        g := new(errgroup.Group)
        g.SetLimit(s.MaxConcurrent)
        for _, t := range tasks {
            t := t
            g.Go(func() error {
                if err := s.runOne(ctx, t, app); err != nil {
                    s.Log.Warn("task_error_best_effort",
                        slog.String("task", t.Name()),
                        slog.String("module", t.Module()),
                        slog.Any("err", err))
                    return nil  // swallow
                }
                return nil
            })
        }
        return g.Wait()
    }

    // runOne handles checkpoint gate + Run + heartbeat + checkpoint Complete around a single task.
    func (s *Scheduler) runOne(ctx context.Context, t task.Task, app any) error {
        // 1. Compute input_hash (Phase 3 plan-04 uses a placeholder cfgSliceJSON; Plan 05 wires real)
        cfgSliceJSON := []byte("{}")  // TODO(plan-05): pull from app.Cfg via task-specific slice
        wordlistsLock := []byte("")    // TODO(plan-05/11): pull from wordlists.lock
        target := "default-target"     // TODO(plan-05): app.Target.Domain
        hash := checkpoint.InputHash(t.Name(), target, cfgSliceJSON, wordlistsLock)

        // 2. Checkpoint Done gate
        if s.Checkpoint != nil {
            done, err := s.Checkpoint.Done(ctx, t.Name(), target, hash)
            if err != nil { return err }
            if done {
                s.Log.Info("task_skipped_checkpoint_hit", slog.String("task", t.Name()))
                return nil
            }
            if err := s.Checkpoint.Begin(ctx, t.Name(), target, hash); err != nil { return err }
        }

        // 3. LifecycleAware.OnStart
        if la, ok := t.(task.LifecycleAware); ok {
            if err := la.OnStart(ctx, app); err != nil {
                s.Log.Warn("task_onstart_failed", slog.String("task", t.Name()), slog.Any("err", err))
            }
        }

        // 4. Heartbeat
        cadence := time.Duration(s.HeartbeatSeconds) * time.Second
        stopHB := startHeartbeat(ctx, t.Name(), t.Module(), cadence)
        defer stopHB()

        // 5. Run
        start := time.Now()
        result, runErr := t.Run(ctx, app)
        result.Duration = time.Since(start)

        // 6. LifecycleAware.OnEnd
        if la, ok := t.(task.LifecycleAware); ok {
            if endErr := la.OnEnd(ctx, app, result); endErr != nil {
                s.Log.Warn("task_onend_failed", slog.String("task", t.Name()), slog.Any("err", endErr))
            }
        }

        // 7. Checkpoint Complete
        if s.Checkpoint != nil {
            _ = s.Checkpoint.Complete(ctx, t.Name(), target, hash, result.Outputs, runErr)
        }

        return runErr
    }
    ```

    ```go
    package lint_test

    import (
        "go/ast"
        "go/parser"
        "go/token"
        "os"
        "path/filepath"
        "strings"
        "testing"
    )

    // Allowlist of file paths (relative to repo root) where exec.Command / exec.CommandContext / os/exec.Cmd{} are permitted.
    // FOUND-10 mandate: every other file calling these is a CI failure.
    var allowedPaths = []string{
        "internal/core/backend/local.go",
        "internal/core/backend/local_test.go",
        "internal/core/backend/local_smoke_test.go",
        "internal/core/backend/registry.go",  // exec.LookPath is fine; check forbids Command/CommandContext only
        "internal/core/backend/registry_test.go",
        // Test infrastructure may also need it; review case-by-case.
    }

    func TestFOUND10_NoRawSubprocessOutsideBackend(t *testing.T) {
        repoRoot, err := findRepoRoot()
        if err != nil { t.Fatalf("cannot find repo root: %v", err) }
        violations := []string{}
        err = filepath.Walk(filepath.Join(repoRoot, "internal"), func(path string, info os.FileInfo, err error) error {
            if err != nil { return err }
            if info.IsDir() { return nil }
            if !strings.HasSuffix(path, ".go") { return nil }
            rel, _ := filepath.Rel(repoRoot, path)
            for _, allowed := range allowedPaths {
                if rel == allowed { return nil }
            }
            // Parse and scan AST for: exec.Command, exec.CommandContext, os/exec.Cmd{} composite literal
            if hit := scanForRawSubprocess(path); hit != "" {
                violations = append(violations, rel+": "+hit)
            }
            return nil
        })
        if err != nil { t.Fatalf("walk: %v", err) }
        // Also scan cmd/, internal/modules/
        for _, base := range []string{"cmd", "internal/modules"} {
            absBase := filepath.Join(repoRoot, base)
            if _, err := os.Stat(absBase); os.IsNotExist(err) { continue }
            filepath.Walk(absBase, func(path string, info os.FileInfo, err error) error {
                if info.IsDir() || !strings.HasSuffix(path, ".go") { return nil }
                rel, _ := filepath.Rel(repoRoot, path)
                if hit := scanForRawSubprocess(path); hit != "" {
                    violations = append(violations, rel+": "+hit)
                }
                return nil
            })
        }
        if len(violations) > 0 {
            t.Errorf("FOUND-10: raw exec.Command forbidden outside internal/core/backend/local.go.\nUse app.Tools.Run() or app.Tools.Stream() instead.\nViolations:\n  %s", strings.Join(violations, "\n  "))
        }
    }

    func scanForRawSubprocess(path string) string {
        fset := token.NewFileSet()
        f, err := parser.ParseFile(fset, path, nil, parser.ParseComments)
        if err != nil { return "" }
        // Skip if file imports nothing called "os/exec" or "exec"
        usesExec := false
        for _, imp := range f.Imports {
            if imp.Path.Value == `"os/exec"` { usesExec = true; break }
        }
        if !usesExec { return "" }
        hit := ""
        ast.Inspect(f, func(n ast.Node) bool {
            if call, ok := n.(*ast.CallExpr); ok {
                if sel, ok := call.Fun.(*ast.SelectorExpr); ok {
                    if pkg, ok := sel.X.(*ast.Ident); ok && pkg.Name == "exec" {
                        if sel.Sel.Name == "Command" || sel.Sel.Name == "CommandContext" {
                            pos := fset.Position(call.Pos())
                            hit = pos.String() + " " + pkg.Name + "." + sel.Sel.Name
                            return false
                        }
                    }
                }
            }
            return true
        })
        return hit
    }

    func findRepoRoot() (string, error) {
        wd, err := os.Getwd()
        if err != nil { return "", err }
        for {
            if _, err := os.Stat(filepath.Join(wd, "go.mod")); err == nil { return wd, nil }
            parent := filepath.Dir(wd)
            if parent == wd { return "", os.ErrNotExist }
            wd = parent
        }
    }
    ```
