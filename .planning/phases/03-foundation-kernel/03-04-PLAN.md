    `(*LocalBackend) Exec`: buffered output; kill-tree pattern from spike `proc.go`:
    1. `cmd := exec.CommandContext(ctx, t.Path, args...)`
    2. `cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}` (process group isolation)
    3. `cmd.WaitDelay = b.KillGrace` (default 5s)
    4. `cmd.Cancel = func() error { if cmd.Process == nil { return nil }; return syscall.Kill(-cmd.Process.Pid, syscall.SIGTERM) }`
    5. Capture stdout/stderr via `cmd.StdoutPipe()` / `cmd.StderrPipe()` into bytes.Buffer
    6. `cmd.Start()`
    7. Launch group-SIGKILL goroutine — when ctx cancelled, sleep `WaitDelay + 500ms`, then `syscall.Kill(-pgid, syscall.SIGKILL)` (ADR-cited workaround for Go's WaitDelay-only-kills-direct-child bug)
    8. `cmd.Wait()` and inspect ExitCode
    9. Return `*Result` on success; `*coreerrors.ToolError{Tool: t.Name, ExitCode: ..., Stderr: lastKB(stderrBuf), Inner: runErr}` on non-zero exit
    10. Return `*coreerrors.ToolTimeout{Tool: t.Name, Timeout: ...}` if ctx deadline exceeded (`errors.Is(ctx.Err(), context.DeadlineExceeded)`)

    `(*LocalBackend) Stream`: streamed output; same kill-tree as Exec but yields Events on a channel:
    1-7. Same as Exec but use both StdoutPipe and StderrPipe
    8. Spawn 2 goroutines (stdout reader + stderr reader) each scanning lines and sending Events on the channel
    9. Use `bufio.NewScanner` with `Buffer(make([]byte, 1024*1024), 10*1024*1024)` (1MiB initial / 10MiB max per RESEARCH.md §Pattern 3)
    10. Close the channel when both readers AND `cmd.Wait()` complete
    11. CRITICAL: caller MUST drain the channel until closed (per ADR §5.2 line 1674-1675); document in godoc

    package backend

    import (
        "context"
        "os/exec"
        "sort"
        "sync"
    )

    type ToolRegistry struct {
        mu      sync.RWMutex
        tools   map[string]*Tool
        missing []string // populated by Discover — all missing-from-PATH tools
    }

    var Default = &ToolRegistry{tools: map[string]*Tool{}}

    func (r *ToolRegistry) Register(t *Tool) {
        r.mu.Lock()
        defer r.mu.Unlock()
        if _, exists := r.tools[t.Name]; exists {
            panic("reconftw: duplicate tool registration: " + t.Name)
        }
        r.tools[t.Name] = t
    }

    func (r *ToolRegistry) Lookup(name string) (*Tool, bool) {
        r.mu.RLock()
        defer r.mu.RUnlock()
        t, ok := r.tools[name]
        return t, ok
    }

    func (r *ToolRegistry) All() []*Tool {
        r.mu.RLock()
        defer r.mu.RUnlock()
        names := make([]string, 0, len(r.tools))
        for n := range r.tools {
            names = append(names, n)
        }
        sort.Strings(names)
        out := make([]*Tool, 0, len(names))
        for _, n := range names {
            out = append(out, r.tools[n])
        }
        return out
    }

    // Discover walks PATH for each registered tool via exec.LookPath; populates .Path.
    // Tools not found are recorded in r.missing for later Health-Check.
    func (r *ToolRegistry) Discover(ctx context.Context) error {
        r.mu.Lock()
        defer r.mu.Unlock()
        r.missing = nil
        for name, t := range r.tools {
            path, err := exec.LookPath(name)
            if err != nil {
                r.missing = append(r.missing, name)
                continue
            }
            t.Path = path
        }
        sort.Strings(r.missing)
        return nil
    }

    // MissingRequired returns the list of registered tools NOT found on PATH (regardless of criticality).
    // FOUND-08: emit structured WARN for these entries.
    func (r *ToolRegistry) MissingRequired() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()
        return append([]string(nil), r.missing...)
    }

    // MissingCritical returns ONLY missing tools whose Critical bool is true.
    // FOUND-08 + Blocker 5: emit structured ERROR for these; health-check exits 1.
    func (r *ToolRegistry) MissingCritical() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()
        out := []string{}
        for _, name := range r.missing {
            if t, ok := r.tools[name]; ok && t.Critical {
                out = append(out, name)
            }
        }
        sort.Strings(out)
        return out
    }
